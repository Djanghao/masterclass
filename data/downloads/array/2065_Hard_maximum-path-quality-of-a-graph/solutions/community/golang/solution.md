// maximalPathQuality builds a weighted, undirected graph from the given
// edges, calculates the shortest paths from the source to each other
// vertex, then uses this data to speed up the DFS calculation of the
// maximum quality.
func maximalPathQuality(values []int, edges [][]int, maxTime int) int {
    g := newGraph(len(values), edges)
    shortestPaths := newDijkstraSP(g, 0)
    return calcMaxQuality(g, shortestPaths, values, maxTime)
}

// Use depth-first search to explore all possible paths of weight maxTime
// that start and end at the source. Each time we encounter the source node,
// we record the quality of the current path and compare it to the maximum
// quality that could be obtained by exploring further. The maximum of these
// values is returned up the stack at each stage, resulting in the highest
// overall quality being returned.
// Using the shortest paths generated by Dijkstra's algorithm, we ensure that
// we only explore paths that have a possibility of ending at the source.
// Time: O(me), where m is the number of paths of weight <= maxTime that start
// and end at the source, and e is the number of edge stemming from the vertices
// on such a path.
// Space: O(p), where p is the maximum length of a path with weight <= maxTime,
// which may be greater than the number of vertices in the graph.
func calcMaxQuality(g graph, shortestPaths *dijkstraSP, values []int, maxTime int) int {
    visited := make([]int, len(g))

    var dfs func(v, time, quality int) int
    dfs = func(v, time, quality int) int {
        // If there's no way to get back to the source in time, this path is no
        // good.
        if time+shortestPaths.timeTo[v] > maxTime {
            return 0
        }

        // Add the value of the current node on the first visit.
        if visited[v] == 0 {
            quality += values[v]
        }

        // Track the number of visits to the node in the current recursion.
        visited[v]++

        // The maxQuality for the current recursion is 0 until the path meets
        // the source again.
        var maxQuality int
        if v == 0 {
            maxQuality = quality
        }

        // Check each neighbor for paths with a higher maxQuality than we've
        // found so far.
        for _, e := range g[v] {
            w := e.other(v)
            timeToW := time + e.time
            qualityThroughW := dfs(w, timeToW, quality)
            maxQuality = max(maxQuality, qualityThroughW)
        }

        visited[v]--
        
        return maxQuality
    }


    return dfs(0, 0, 0)
}

type edge struct {
    v, w int
    time int
}

func (e *edge) either() int {
    return e.v
}

func (e *edge) other(x int) int {
    if x == e.v {
        return e.w
    }
    if x == e.w {
        return e.v
    }
    panic(errors.New(fmt.Sprintf("edge %+v has no vertex %d", e, x)))
}

type graph [][]*edge

// newGraph builds a graph from the given edges.
func newGraph(n int, edges [][]int) graph {
    g := make(graph, n)
    for _, inputEdge := range edges {
        v, w, time := inputEdge[0], inputEdge[1], inputEdge[2]
        graphEdge := &edge{v, w, time}
        g[v] = append(g[v], graphEdge)
        g[w] = append(g[w], graphEdge)
    }

    return g
}

// dijkstraSP calculates and stores the smallest time to travel from
// the given source node to each other node in the graph using Dijkstra's
// algorithm.
type dijkstraSP struct {
    timeTo []int
}

func newDijkstraSP(g graph, source int) *dijkstraSP {
    d := &dijkstraSP{}
    d.build(g, source)
    return d
}

// Dijkstra's algorithm. This implementation uses a symbol heap with
// random access to store only the shortest known edge to each vertex
// whose shortest path has not yet been discovered. This allows us
// to calculate all shortest paths in O(E\xB7logV) time.
func (d *dijkstraSP) build(g graph, source int) {
    timeTo := make([]int, len(g))
    timePQ := newTimeMinHeap()

    for i := range timeTo {
        timeTo[i] = math.MaxInt
    }
    timeTo[source] = 0
    timePQ.Push(0, 0)

    relax := func(v int) {
        for _, e := range g[v] {
            w := e.other(v)
            candidateTime := timeTo[v] + e.time
            if candidateTime < timeTo[w] {
                timeTo[w] = candidateTime
                timePQ.Update(w, candidateTime)
            }
        }
    }

    for v, _, ok := timePQ.Pop(); ok; v, _, ok = timePQ.Pop() {
        relax(v)
    }

    d.timeTo = timeTo
}

// timeMinHeap implements a priority queue for integers with random access
// using integer keys that correspond to vertices. The nature of Go's
// standard container/heap interface is such that it requires just as many
// lines of code to implement a symbol heap from scratch as it does to
// wrap the standard library functions with extra logic to handle random
// access.
type timeMinHeap struct {
    data []int
    indices map[int]int
    keys []int
}

func newTimeMinHeap() *timeMinHeap {
    return &timeMinHeap {
        indices: make(map[int]int),
    }
}

func (h *timeMinHeap) Contains(key int) bool {
    _, ok := h.indices[key]
    return ok
}

func (h *timeMinHeap) Push(k int, v int) {
    if h.Contains(k) {
        h.update(k, v)
        return
    }
    h.push(k, v)
}

func (h *timeMinHeap) push(k int, v int) {
    h.data = append(h.data, v)
    h.keys = append(h.keys, k)
    h.indices[k] = len(h.data)-1

    h.swim(len(h.data)-1)
}

func (h *timeMinHeap) Update(k int, v int) {
    if !h.Contains(k) {
        h.push(k, v)
        return
    }
    h.update(k, v)
}

func (h *timeMinHeap) update(k int, v int) {
    idx := h.indices[k]
    h.data[idx] = v
    h.swim(idx)
    h.sink(idx)
}

func (h *timeMinHeap) Pop() (int, int, bool) {
    if len(h.data) == 0 {
        return 0, 0, false
    }

    k, v := h.keys[0], h.data[0]
    lastIdx := len(h.data)-1
    h.swap(0, lastIdx)
    h.data[lastIdx] = 0
    h.data = h.data[:lastIdx]
    h.keys = h.keys[:lastIdx]
    delete(h.indices, k)
    h.sink(0)

    return k, v, true
}

func (h *timeMinHeap) swim(idx int) {
    for ; idx > 0 && h.hasPriority(idx, (idx-1)/2); idx = (idx-1)/2 {
        h.swap(idx, (idx-1)/2)
    }
}

func (h *timeMinHeap) sink(idx int) {
    for 2*idx+1 < len(h.data) {
        priorityChild := 2*idx + 1
        if priorityChild < len(h.data)-1 && h.hasPriority(priorityChild+1, priorityChild) {
            priorityChild++
        }

        if !h.hasPriority(priorityChild, idx) {
            break
        }

        h.swap(idx, priorityChild)
        idx = priorityChild
    }
}

func (h *timeMinHeap) hasPriority(i, j int) bool {
    return h.data[i] < h.data[j]
} 

func (h *timeMinHeap) swap(i, j int) {
    iKey, jKey := h.keys[i], h.keys[j]
    h.data[i], h.data[j] = h.data[j], h.data[i]
    h.keys[i], h.keys[j] = h.keys[j], h.keys[i]
    h.indices[iKey], h.indices[jKey] = h.indices[jKey], h.indices[iKey]
}

func max(a, b int) int {
    if a > b {
        return a
    }
    return b
}
