 class Solution {

    /**
     * \u0424\u0443\u043D\u043A\u0446\u0438\u044F maximumScore \u043F\u0440\u0438\u043D\u0438\u043C\u0430\u0435\u0442 \u043E\u0446\u0435\u043D\u043A\u0438 \u0443\u0437\u043B\u043E\u0432 $scores \u0438 \u043C\u0430\u0441\u0441\u0438\u0432 \u0440\u0451\u0431\u0435\u0440 $edges.
     * \u0412\u043E\u0437\u0432\u0440\u0430\u0449\u0430\u0435\u0442 \u043C\u0430\u043A\u0441\u0438\u043C\u0430\u043B\u044C\u043D\u0443\u044E \u0441\u0443\u043C\u043C\u0443 \u043E\u0446\u0435\u043D\u043E\u043A \u0432\u0430\u043B\u0438\u0434\u043D\u043E\u0439 \u043F\u043E\u0441\u043B\u0435\u0434\u043E\u0432\u0430\u0442\u0435\u043B\u044C\u043D\u043E\u0441\u0442\u0438 \u0438\u0437 4 \u0443\u0437\u043B\u043E\u0432.
     *
     * @param Integer[] $scores \u041C\u0430\u0441\u0441\u0438\u0432 \u043E\u0446\u0435\u043D\u043E\u043A \u0434\u043B\u044F \u043A\u0430\u0436\u0434\u043E\u0433\u043E \u0443\u0437\u043B\u0430.
     * @param Integer[][] $edges \u041C\u0430\u0441\u0441\u0438\u0432 \u0440\u0451\u0431\u0435\u0440 \u0432 \u0433\u0440\u0430\u0444\u0435.
     * @return Integer \u041C\u0430\u043A\u0441\u0438\u043C\u0430\u043B\u044C\u043D\u0430\u044F \u0441\u0443\u043C\u043C\u0430 \u043E\u0446\u0435\u043D\u043E\u043A \u0432\u0430\u043B\u0438\u0434\u043D\u043E\u0439 \u043F\u043E\u0441\u043B\u0435\u0434\u043E\u0432\u0430\u0442\u0435\u043B\u044C\u043D\u043E\u0441\u0442\u0438 \u0438\u0437 4 \u0443\u0437\u043B\u043E\u0432.
     */
    function maximumScore($scores, $edges) {
        // \u041F\u043E\u043B\u0443\u0447\u0435\u043D\u0438\u0435 \u043A\u043E\u043B\u0438\u0447\u0435\u0441\u0442\u0432\u0430 \u0443\u0437\u043B\u043E\u0432 \u0432 \u0433\u0440\u0430\u0444\u0435
        $n = count($scores);
        
        // \u0421\u043E\u0437\u0434\u0430\u043D\u0438\u0435 \u043C\u0430\u0441\u0441\u0438\u0432\u0430 \u0441\u043C\u0435\u0436\u043D\u043E\u0441\u0442\u0438 \u0434\u043B\u044F \u0445\u0440\u0430\u043D\u0435\u043D\u0438\u044F \u0441\u043E\u0441\u0435\u0434\u0435\u0439 \u043A\u0430\u0436\u0434\u043E\u0433\u043E \u0443\u0437\u043B\u0430
        $adj = array_fill(0, $n, []);

        // \u0417\u0430\u043F\u043E\u043B\u043D\u0435\u043D\u0438\u0435 \u043C\u0430\u0441\u0441\u0438\u0432\u0430 \u0441\u043C\u0435\u0436\u043D\u043E\u0441\u0442\u0438 \u043D\u0430 \u043E\u0441\u043D\u043E\u0432\u0435 \u043C\u0430\u0441\u0441\u0438\u0432\u0430 \u0440\u0451\u0431\u0435\u0440
        foreach ($edges as $edge) {
            $adj[$edge[0]][] = $edge[1]; // \u0414\u043E\u0431\u0430\u0432\u043B\u0435\u043D\u0438\u0435 \u0432\u0435\u0440\u0448\u0438\u043D\u044B $edge[1] \u0432 \u0441\u043F\u0438\u0441\u043E\u043A \u0441\u043E\u0441\u0435\u0434\u0435\u0439 \u0432\u0435\u0440\u0448\u0438\u043D\u044B $edge[0]
            $adj[$edge[1]][] = $edge[0]; // \u0414\u043E\u0431\u0430\u0432\u043B\u0435\u043D\u0438\u0435 \u0432\u0435\u0440\u0448\u0438\u043D\u044B $edge[0] \u0432 \u0441\u043F\u0438\u0441\u043E\u043A \u0441\u043E\u0441\u0435\u0434\u0435\u0439 \u0432\u0435\u0440\u0448\u0438\u043D\u044B $edge[1]
        }

        // \u0414\u043B\u044F \u043A\u0430\u0436\u0434\u043E\u0439 \u0432\u0435\u0440\u0448\u0438\u043D\u044B \u0441\u043E\u0437\u0434\u0430\u0451\u0442\u0441\u044F \u043F\u0440\u0438\u043E\u0440\u0438\u0442\u0435\u0442\u043D\u0430\u044F \u043E\u0447\u0435\u0440\u0435\u0434\u044C \u0441 3 \u043D\u0430\u0438\u0432\u044B\u0441\u0448\u0438\u043C\u0438 \u043E\u0446\u0435\u043D\u043A\u0430\u043C\u0438 \u0441\u043E\u0441\u0435\u0434\u0435\u0439
        foreach ($adj as &$neighbors) {
            $pq = new SplPriorityQueue();
            foreach ($neighbors as $vertex) {
                $score = $scores[$vertex];
                // \u0412\u0441\u0442\u0430\u0432\u043A\u0430 \u043E\u0446\u0435\u043D\u043A\u0438 \u0432\u0435\u0440\u0448\u0438\u043D\u044B \u0432 \u043F\u0440\u0438\u043E\u0440\u0438\u0442\u0435\u0442\u043D\u0443\u044E \u043E\u0447\u0435\u0440\u0435\u0434\u044C \u0441 \u043E\u0442\u0440\u0438\u0446\u0430\u0442\u0435\u043B\u044C\u043D\u044B\u043C \u043F\u0440\u0438\u043E\u0440\u0438\u0442\u0435\u0442\u043E\u043C
                $pq->insert($vertex, -$score);
                if ($pq->count() > 3) {
                    $pq->extract(); // \u0423\u0434\u0430\u043B\u0435\u043D\u0438\u0435 \u043D\u0430\u0438\u043C\u0435\u043D\u044C\u0448\u0435\u0439 \u043E\u0446\u0435\u043D\u043A\u0438
                }
            }
            $neighbors = [];
            // \u0418\u0437\u0432\u043B\u0435\u0447\u0435\u043D\u0438\u0435 \u0434\u0430\u043D\u043D\u044B\u0445 \u0438\u0437 \u043F\u0440\u0438\u043E\u0440\u0438\u0442\u0435\u0442\u043D\u043E\u0439 \u043E\u0447\u0435\u0440\u0435\u0434\u0438 \u0432 \u043C\u0430\u0441\u0441\u0438\u0432 \u0441\u043E\u0441\u0435\u0434\u0435\u0439
            $pq->setExtractFlags(SplPriorityQueue::EXTR_DATA);
            while (!$pq->isEmpty()) {
                $neighbors[] = $pq->extract();
            }
        }

        // \u0418\u043D\u0438\u0446\u0438\u0430\u043B\u0438\u0437\u0430\u0446\u0438\u044F \u043F\u0435\u0440\u0435\u043C\u0435\u043D\u043D\u043E\u0439 \u0434\u043B\u044F \u0445\u0440\u0430\u043D\u0435\u043D\u0438\u044F \u043C\u0430\u043A\u0441\u0438\u043C\u0430\u043B\u044C\u043D\u043E\u0439 \u0441\u0443\u043C\u043C\u044B \u043E\u0446\u0435\u043D\u043E\u043A \u043F\u043E\u0441\u043B\u0435\u0434\u043E\u0432\u0430\u0442\u0435\u043B\u044C\u043D\u043E\u0441\u0442\u0438 \u0438\u0437 4 \u0443\u0437\u043B\u043E\u0432
        $result = -1;

        // \u041F\u0435\u0440\u0435\u0431\u043E\u0440 \u0432\u0441\u0435\u0445 \u0440\u0451\u0431\u0435\u0440 \u0432 \u0433\u0440\u0430\u0444\u0435
        foreach ($edges as $edge) {
            $u = $edge[0];
            $v = $edge[1];
            $base = $scores[$u] + $scores[$v];

            // \u041F\u0435\u0440\u0435\u0431\u043E\u0440 \u0441\u043E\u0441\u0435\u0434\u0435\u0439 \u0432\u0435\u0440\u0448\u0438\u043D, \u0441\u043E\u0435\u0434\u0438\u043D\u0451\u043D\u043D\u044B\u0445 \u0442\u0435\u043A\u0443\u0449\u0438\u043C \u0440\u0435\u0431\u0440\u043E\u043C
            foreach ($adj[$u] as $x) {
                if ($x === $v) continue;
                foreach ($adj[$v] as $y) {
                    if ($y === $u || $x === $y) continue;
                    // \u041F\u043E\u0438\u0441\u043A \u043C\u0430\u043A\u0441\u0438\u043C\u0430\u043B\u044C\u043D\u043E\u0439 \u0441\u0443\u043C\u043C\u044B \u043E\u0446\u0435\u043D\u043E\u043A \u043F\u043E\u0441\u043B\u0435\u0434\u043E\u0432\u0430\u0442\u0435\u043B\u044C\u043D\u043E\u0441\u0442\u0438 \u0438\u0437 4 \u0443\u0437\u043B\u043E\u0432
                    $result = max($result, $base + $scores[$x] + $scores[$y]);
                }
            }
        }

        // \u0412\u043E\u0437\u0432\u0440\u0430\u0449\u0430\u0435\u043C \u043C\u0430\u043A\u0441\u0438\u043C\u0430\u043B\u044C\u043D\u0443\u044E \u0441\u0443\u043C\u043C\u0443 \u043E\u0446\u0435\u043D\u043E\u043A
        return $result;
    }
}
