# Intuition
The goal is to find the closest value to a given target using bitwise operations. Each number in the array can combine with others using the AND operation, potentially creating new values that might be closer to the target.

# Approach
Use a sorted structure to efficiently store and check new values generated by ANDing each number with existing values while updating the closest difference to the target.

# Complexity
- Time complexity:
O(n2) because we iterate through the array and check combinations for each element.

- Space complexity:
O(n) since we store unique values in a sorted structure.

// A utility class for maintaining sorted arrays with binary search operations
class Bisect {
    public insort_right(a: number[], x: number, lo: number = 0, hi: number | null = null): void {
        lo = this.bisect_right(a, x, lo, hi);
        a.splice(lo, 0, x); // Insert x at position lo
    }

    public bisect_right(a: number[], x: number, lo: number = 0, hi: number | null = null): number {
        if (lo < 0) throw new Error('lo must be non-negative');
        if (hi === null) hi = a.length; // If hi is not set, use the length of the array

        while (lo < hi) {
            const mid = Math.floor((lo + hi) / 2); // Calculate mid index
            a[mid] > x ? (hi = mid) : (lo = mid + 1); // Adjust search bounds
        }
        return lo; // Return the index for insertion
    }

    public insort_left(a: number[], x: number, lo: number = 0, hi: number | null = null): void {
        lo = this.bisect_left(a, x, lo, hi);
        a.splice(lo, 0, x); // Insert x at position lo
    }

    public bisect_left(a: number[], x: number, lo: number = 0, hi: number | null = null): number {
        if (lo < 0) throw new Error('lo must be non-negative');
        if (hi === null) hi = a.length; // If hi is not set, use the length of the array

        while (lo < hi) {
            const mid = Math.floor((lo + hi) / 2); // Calculate mid index
            a[mid] < x ? (lo = mid + 1) : (hi = mid); // Adjust search bounds
        }
        return lo; // Return the index for insertion
    }
}

// A class representing a TreeSet using a sorted array and a Set for uniqueness
class TreeSet {
    private ts: number[]; // Sorted array to maintain elements
    private se: Set<number>; // Set for uniqueness
    private bisect: Bisect; // Instance of Bisect for binary search operations

    constructor(elements?: number[]) {
        this.ts = [];
        this.se = new Set();
        this.bisect = new Bisect();
        this.initialize(elements);
    }

    // Initialize the TreeSet with provided elements
    private initialize(elements?: number[]) {
        if (elements) {
            for (const e of elements) {
                if (!this.se.has(e)) {
                    this.bisect.insort_right(this.ts, e);
                    this.se.add(e);
                }
            }
        }
    }

    // Add an element to the TreeSet
    public add(e: number): void {
        if (!this.se.has(e)) {
            this.bisect.insort_right(this.ts, e);
            this.se.add(e);
        }
    }

    // Get the first element
    public first(): number | undefined {
        return this.ts[0];
    }

    // Get the last element
    public last(): number | undefined {
        return this.ts[this.ts.length - 1];
    }

    // Remove and return the first element
    public poll(): number | undefined {
        const res = this.ts.shift(); // Remove the first element
        if (res !== undefined) this.se.delete(res); // Remove from the set
        return res;
    }

    // Remove and return the last element
    public pollLast(): number | undefined {
        const res = this.ts.pop(); // Remove the last element
        if (res !== undefined) this.se.delete(res); // Remove from the set
        return res;
    }

    // Find the smallest element greater than or equal to e
    public ceiling(e: number): number | null {
        const idx = this.bisect.bisect_right(this.ts, e);
        const res = (idx > 0 && this.ts[idx - 1] === e) ? e : this.ts[idx];
        return res === undefined ? null : res;
    }

    // Find the smallest element greater than e
    public higher(e: number): number | null {
        const idx = this.bisect.bisect_right(this.ts, e);
        const res = (idx < this.ts.length) ? this.ts[idx] : null;
        return res === undefined ? null : res;
    }

    // Find the largest element less than or equal to e
    public floor(e: number): number | null {
        const idx = this.bisect.bisect_left(this.ts, e);
        const res = (idx > 0 && this.ts[idx - 1] === e) ? e : this.ts[idx - 1];
        return res === undefined ? null : res;
    }

    // Find the largest element less than e
    public lower(e: number): number | null {
        const idx = this.bisect.bisect_left(this.ts, e);
        const res = (idx > 0) ? this.ts[idx - 1] : null;
        return res === undefined ? null : res;
    }

    // Remove an element from the TreeSet
    public remove(e: number): void {
        const idx = this.bisect.bisect_left(this.ts, e);
        if (this.ts[idx] === e) {
            this.ts.splice(idx, 1); // Remove from sorted array
        }
        this.se.delete(e); // Remove from the set
    }

    // Check if the TreeSet contains an element
    public contains(e: number): boolean {
        return this.se.has(e);
    }

    // Get the size of the TreeSet
    public size(): number {
        return this.ts.length;
    }

    // Clear all elements from the TreeSet
    public clear(): void {
        this.ts = [];
        this.se.clear();
    }

    // Show all elements in the TreeSet
    public show(): number[] {
        return this.ts;
    }
}

// Function to find the closest value to target using bitwise AND operation
const closestToTarget = (a: number[], t: number): number => {
    let ts = new TreeSet();
    let res = Number.MAX_SAFE_INTEGER; // Initialize to the maximum safe integer

    for (const x of a) {
        let nts = new TreeSet();
        nts.add(x); // Start a new TreeSet with the current element

        // Compute AND with existing elements and store results in nts
        for (const y of ts.show()) {
            nts.add(x & y);
        }

        // Update the result with the minimum difference from the target
        for (const y of nts.show()) {
            res = Math.min(res, Math.abs(y - t));
        }

        ts = nts; // Move to the new TreeSet for the next iteration
    }

    return res; // Return the minimum difference found
};
