class Solution {
  List<int> fullBloomFlowers(List<List<int>> flowers, List<int> people) {
      final f = <List<int>>[];
      final p = <List<int>>[];
      final res = List<int>.filled(people.length, 0);
      // f: [i\uBC88\uC9F8 \uC774\uBCA4\uD2B8\uAC00 \uC2E4\uD589\uB418\uB294 \uC2DC\uAC04, 1(\uAF43\uC774 \uD540\uB2E4) or -1(\uAF43\uC774 \uC9C4\uB2E4)]
      for (final flower in flowers){
          f.add([flower[0], 1]);
          f.add([flower[1] + 1, -1]);
      }
      // p: [i\uBC88\uC9F8 \uC0AC\uB78C\uC774 \uB3C4\uCC29\uD558\uB294\uC2DC\uAC04, i\uBC88\uC9F8 \uC0AC\uB78C\uC758 \uC6D0\uB798 index(res\uC5D0 \uB2F4\uC744 \uC21C\uC11C)]
      for (int i = 0; i < people.length; i++){
          p.add([people[i], i]);
      }
      // 1. \uC704\uC5D0\uC11C \uAD6C\uD55C f, p\uB97C \uC2DC\uAC04\uC744 \uAE30\uC900\uC73C\uB85C \uC815\uB82C\uD574\uC90D\uB2C8\uB2E4
      f.sort((a, b) => a[0].compareTo(b[0]));
      p.sort((a, b) => a[0].compareTo(b[0]));

      int cnt = 0;
      int fi = 0;
      // 2. p\uB97C \uC0AC\uB78C\uB4E4\uC774 \uB3C4\uCC29\uD558\uB294 \uC2DC\uAC04\uC744 \uAE30\uC900\uC73C\uB85C \uC21C\uD68C\uD569\uB2C8\uB2E4
      for (int i = 0; i < p.length; i++){
          final pi = p[i][0]; // \uC774 \uC0AC\uB78C\uC774 \uB3C4\uCC29\uD55C \uC2DC\uAC04
          final ri = p[i][1]; // \uC774 \uC0AC\uB78C\uC758 \uC6D0\uB798 people \uB0B4\uC758 index
          // f\uB97C while\uB85C \uC21C\uD68C\uD558\uBA70 2\uBC88\uC758 \uC2DC\uC810 \uC774\uC804\uC5D0 \uBC1C\uC0DD\uD588\uC5B4\uC57C \uD558\uB294 \uC774\uBCA4\uD2B8\uB97C \uBAA8\uB450 \uBC1C\uC0DD\uC2DC\uD0B5\uB2C8\uB2E4.
          while (f.length > fi && f[fi][0] <= pi){
              cnt += f[fi][1];
              fi++;
          }
          // 2\uBC88\uC2DC\uC810\uC5D0 \uD53C\uC5B4\uC788\uB294 \uAF43\uC744 \uD574\uB2F9 \uC2DC\uC810\uC5D0 \uB3C4\uCC29\uD558\uB294 \uC0AC\uB78C\uC758 people \uB0B4\uC758 \uC6D0\uB798 index\uC5D0 \uB2F4\uC544\uC90D\uB2C8\uB2E4
          res[ri] = cnt;
      }
      //  \uB05D
      return res;
  }
}
