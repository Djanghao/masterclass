class Solution {
    func largestTimeFromDigits(_ A: [Int]) -> String {
        guard A.count == 4 else {
            return ""
        }
        
        for i in (0...23).reversed() {
            let h = i % 10
            let dh = i / 10
            if A.contains(dh), A.contains(h) {
                if let result = pickTime(A, h, dh) {
                    return result
                }
            }
        }
        return ""
        
    }
    
    func pickTime(_ from: [Int], _ valid: inout [Int]) -> Int? {
        for i in from {
            if valid.contains(i) {
                if let index = valid.firstIndex(of: i) {
                    valid.remove(at: index)
                }
                return i
            }
        }
        return nil
        
    }
    
    func pickTime(_ from: [Int], _ h: Int, _ dh: Int) -> String? {
        var numbers = from
        if let index = numbers.firstIndex(of: h) {
            numbers.remove(at: index)
        }
        if let index = numbers.firstIndex(of: dh) {
            numbers.remove(at: index)
        } else {
            return nil
        }
        guard let dm = pickTime([5,4,3,2,1,0], &numbers) else {
            return nil
        }
        let m = numbers.last!
        let result = "\\(dh)\\(h):\\(dm)\\(m)"
        
        
        return result
        
    }
}
