// Heap Node.
type Node struct {
	// \u0421\u043E\u0434\u0435\u0440\u0436\u0438\u043C\u043E\u0435 \u0443\u0437\u043B\u0430.
	x string
	// \u041A\u043B\u044E\u0447
	key int
	// \u041F\u0440\u0435\u0434\u043E\u043A \u0443\u0437\u043B\u0430
	parent *Node
	// \u041B\u0435\u0432\u044B\u0439 \u0431\u0440\u0430\u0442\u0441\u043A\u0438\u0439 / \u0441\u0435\u0441\u0442\u0440\u0438\u043D\u0441\u043A\u0438\u0439 \u0443\u0437\u0435\u043B
	left *Node
	// \u041F\u0440\u0430\u0432\u044B\u0439 \u0431\u0440\u0430\u0442\u0441\u043A\u0438\u0439 / \u0441\u0435\u0441\u0442\u0440\u0438\u043D\u0441\u043A\u0438\u0439 \u0443\u0437\u0435\u043B
	right *Node
	// \u041F\u0440\u044F\u043C\u043E\u0439 \u043F\u043E\u0442\u043E\u043C\u043E\u043A \u0443\u0437\u043B\u0430
	child *Node
	// \u0420\u0430\u043D\u0433 \u0443\u0437\u043B\u0430 = \u043A\u043E\u043B-\u0432\u043E \u043F\u0440\u044F\u043C\u044B\u0445 \u043F\u043E\u0442\u043E\u043C\u043A\u043E\u0432
	rank int
	// \u041F\u0435\u0440\u0435\u043C\u0435\u0449\u0430\u043B\u0438\u0441\u044C \u043B\u0438 \u0440\u0430\u043D\u0435\u0435 \u043F\u043E\u0442\u043E\u043C\u043A\u0438 \u044D\u0442\u043E\u0433\u043E \u0443\u0437\u043B\u0430
	marked bool
}

func ConstructNode(x string, key int) Node {
	// Node initialization.
	this := Node{
		x:   x,
		key: key,
	}
	return this
}

func (this *Node) _extract() {
	// \u0423\u0434\u0430\u043B\u0435\u043D\u0438\u0435 \u0441\u0432\u044F\u0437\u0435\u0439 \u043F\u0435\u0440\u0435\u0434 \u043F\u0435\u0440\u0435\u043D\u043E\u0441\u043E\u043C \u0443\u0437\u043B\u0430.
	this.parent = nil
	this.left = nil
	this.right = nil
}

func (this *Node) __repr__() string {
	// Node representation.
	return fmt.Sprintf("Node(x={%d})", this.x)
}

// \u0424\u0438\u0431\u043E\u043D\u0430\u0447\u0447\u0438\u0435\u0432\u0430 \u043A\u0443\u0447\u0430.
type FibonacciHeap struct {
	minNode *Node
}

func ConstructHeap(node *Node) *FibonacciHeap {
	/*
	   \u0421\u043E\u0437\u0434\u0430\u043D\u0438\u0435 \u043D\u043E\u0432\u043E\u0439 \u0444\u0438\u0431\u043E\u043D\u0430\u0447\u0447\u0438\u0435\u0432\u043E\u0439 \u043A\u0443\u0447\u0438.

	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(1)
	*/
	this := &FibonacciHeap{minNode: node}
	return this
}

func (this *FibonacciHeap) insert(node *Node) {
	/*
	   \u0412\u0441\u0442\u0430\u0432\u043A\u0430 \u0443\u0437\u043B\u0430 node \u0432 \u0441\u043F\u0438\u0441\u043E\u043A \u043A\u043E\u0440\u043D\u0435\u0432\u044B\u0445 \u0443\u0437\u043B\u043E\u0432.

	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(1)
	*/
	h2 := ConstructHeap(nil)
	h2._set_min(node)
	this.meld(h2)
}

func (this *FibonacciHeap) _set_min(node *Node) {
	/*
	   \u0423\u0441\u0442\u0430\u043D\u043E\u0432\u043A\u0430 \u043C\u0438\u043D\u0438\u043C\u0430\u043B\u044C\u043D\u043E\u0433\u043E \u0443\u0437\u043B\u0430.

	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(1)
	*/
	this.minNode = node
}

func (this *FibonacciHeap) _update_min(node *Node) {
	/*
	   \u041E\u0431\u043D\u043E\u0432\u043B\u0435\u043D\u0438\u0435 \u043C\u0438\u043D\u0438\u043C\u0430\u043B\u044C\u043D\u043E\u0433\u043E \u0443\u0437\u043B\u0430, \u0435\u0441\u043B\u0438 \u043A\u043B\u044E\u0447 \u043C\u0435\u043D\u044C\u0448\u0435.

	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(1)
	*/
	current := this.find_min()
	if current == nil {
		this._set_min(node)
	} else if node != nil && node.key <= current.key {
		this._set_min(node)
	}
}

func (this *FibonacciHeap) find_min() *Node {
	/*
	   \u041F\u043E\u0438\u0441\u043A \u043C\u0438\u043D\u0438\u043C\u0430\u043B\u044C\u043D\u043E\u0433\u043E \u0443\u0437\u043B\u0430.

	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(1)
	*/
	return this.minNode
}

func (this *FibonacciHeap) meld(h *FibonacciHeap) {
	/*
	   \u041E\u0431\u044A\u0435\u0434\u0438\u043D\u0435\u043D\u0438\u0435 \u0434\u0432\u0443\u0445 \u0444\u0438\u0431\u043E\u043D\u0430\u0447\u0447\u0438\u0435\u0432\u044B\u0445 \u043A\u0443\u0447.

	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(1)
	*/
	node1 := this.find_min()
	node2 := h.find_min()
	// \u0421\u043A\u043B\u0435\u0438\u0432\u0430\u043D\u0438\u0435 \u0434\u0432\u0443\u0445 \u0434\u0432\u0443\u0441\u0432\u044F\u0437\u043D\u044B\u0445 \u0441\u043F\u0438\u0441\u043A\u043E\u0432 (\u043A\u043E\u043B\u0435\u0446)
	// x - \u0443\u0434\u0430\u043B\u044F\u0435\u043C\u0430\u044F \u0441\u0432\u044F\u0437\u044C
	// left1 <-x node1 -> right1
	//       X
	// left2 <-x node2 -> right2

	// \u0414\u043E\u0431\u0430\u0432\u043B\u044F\u0435\u043C\u0430\u044F \u043A\u0443\u0447\u0430 \u043F\u0443\u0441\u0442\u0430
	if node2 == nil {
		return
	}

	// \u0418\u0441\u0445\u043E\u0434\u043D\u0430\u044F \u043A\u0443\u0447\u0430 \u043F\u0443\u0441\u0442\u0430
	if node1 == nil {
		this._set_min(node2)
		return
	}

	// \u041F\u043E\u0441\u043A\u043E\u043B\u044C\u043A\u0443 \u0441\u043F\u0438\u0441\u043E\u043A \u0434\u0432\u0443\u0441\u0432\u044F\u0437\u043D\u044B\u0439 \u043A\u043E\u043B\u044C\u0446\u0435\u0432\u043E\u0439, \u0442\u043E \u0435\u0441\u043B\u0438 \u0435\u0441\u0442\u044C \u043B\u0435\u0432\u044B\u0439 \u0443\u0437\u0435\u043B,
	// \u0442\u043E \u0441\u0443\u0449\u0435\u0441\u0442\u0432\u0443\u0435\u0442 \u043F\u0440\u0430\u0432\u044B\u0439 (\u0440\u0430\u0432\u0435\u043D \u043B\u0435\u0432\u043E\u043C\u0443 \u0438\u043B\u0438 \u0434\u0440\u0443\u0433\u043E\u043C\u0443)
	// \u0415\u0441\u043B\u0438 \u0432 \u0441\u043F\u0438\u0441\u043A\u0435 1 \u044D\u043B\u0435\u043C\u0435\u043D\u0442, \u0442\u043E \u043E\u043D \u043D\u0435 \u0443\u043A\u0430\u0437\u044B\u0432\u0430\u0435\u0442 \u0441\u0430\u043C \u043D\u0430 \u0441\u0435\u0431\u044F = None
	left1 := node1.left
	left2 := node2.left

	// \u0412 \u0438\u0441\u0445\u043E\u0434\u043D\u043E\u0439 \u043A\u0443\u0447\u0435 1 \u043A\u043E\u0440\u043D\u0435\u0432\u043E\u0439 \u0443\u0437\u0435\u043B
	if left1 == nil {
		if left2 != nil {
			// \u041F\u043E \u043B\u0435\u0432\u043E\u043C\u0443 \u0443\u0437\u043B\u0443 \u0432\u0442\u043E\u0440\u043E\u0439 \u043A\u0443\u0447\u0438
			//      node1
			//   |        |
			// left2 <-x node2
			node1.left = left2
			node1.right = node2
			left2.right = node1
			node2.left = node1
		} else {
			// \u0412 \u043E\u0431\u0435\u0438\u0445 \u043A\u0443\u0447\u0430\u0445 1 \u043A\u043E\u0440\u043D\u0435\u0432\u043E\u0439 \u0443\u0437\u0435\u043B
			// node1
			//   |
			// node2
			node1.left = node2
			node1.right = node2
			node2.left = node1
			node2.right = node1
		}
	} else {
		// \u0421\u043A\u043B\u0435\u0438\u0432\u0430\u0435\u043C \u0447\u0435\u0440\u0435\u0437 \u043B\u0435\u0432\u044B\u0439 \u043A\u043E\u0440\u043D\u0435\u0432\u043E\u0439 \u0443\u0437\u0435\u043B \u0432\u0442\u043E\u0440\u043E\u0439 \u043A\u0443\u0447\u0438
		if left2 != nil {
			// left1 <-x node1
			//        X
			// left2 <-x node2
			// \u043D\u0430\u0438\u0441\u043A\u043E\u0441\u043E\u043A
			left1.right = node2
			node1.left = left2
			left2.right = node1
			node2.left = left1
			// \u0412\u043E \u0432\u0442\u043E\u0440\u043E\u0439 \u043A\u0443\u0447\u0435 1 \u043A\u043E\u0440\u043D\u0435\u0432\u043E\u0439 \u0443\u0437\u0435\u043B
		} else {
			// left1 <-x node1
			//   |        |
			//      node2
			node2.left = left1
			node2.right = node1
			left1.right = node2
			node1.left = node2
		}
	}
	// \u0415\u0441\u043B\u0438 \u043D\u0443\u0436\u043D\u043E, \u043E\u0431\u043D\u043E\u0432\u043B\u044F\u0435\u043C \u043C\u0438\u043D\u0438\u043C\u0443\u043C
	this._update_min(node2)
}

func (this *FibonacciHeap) delete_min() *Node {
	/*
	   \u0418\u0437\u0432\u043B\u0435\u0447\u0435\u043D\u0438\u0435 \u043C\u0438\u043D\u0438\u043C\u0430\u043B\u044C\u043D\u043E\u0433\u043E \u0443\u0437\u043B\u0430.

	   	x
	   	/ | \\
	   c1 c2 c3
	   \u0410\u043C\u043E\u0440\u0442\u0438\u0437\u0438\u0440\u043E\u0432\u0430\u043D\u043D\u043E\u0435 \u0432\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(log n)
	*/
	root := this.find_min()
	if root == nil {
		// raise ValueError("\u041A\u0443\u0447\u0430 \u043F\u0443\u0441\u0442\u0430")
		return nil
	}
	// \u0423\u0441\u0442\u0430\u043D\u0430\u0432\u043B\u0438\u0432\u0430\u0435\u043C \u0432\u0440\u0435\u043C\u0435\u043D\u043D\u043E \u043C\u0438\u043D\u0438\u043C\u0430\u043B\u044C\u043D\u044B\u0439 \u0443\u0437\u0435\u043B \u043D\u0430 \u043B\u0435\u0432\u044B\u0439
	this._set_min(root.left)
	// \u0423\u0434\u0430\u043B\u044F\u0435\u043C \u0438\u0437 \u0441\u043F\u0438\u0441\u043A\u0430 \u043C\u0438\u043D\u0438\u043C\u0430\u043B\u044C\u043D\u044B\u0439 \u0443\u0437\u0435\u043B
	this._unlink(root)
	// \u0421\u043E\u0437\u0434\u0430\u0435\u043C \u043D\u043E\u0432\u0443\u044E \u043A\u0443\u0447\u0443 \u0438\u0437 \u043F\u043E\u0442\u043E\u043C\u043A\u043E\u0432 root (\u0443 \u043D\u0438\u0445 \u043F\u0440\u0435\u0436\u043D\u0438\u0439 parent)
	h := ConstructHeap(root.child)
	this.meld(h)
	this._consolidate()
	root._extract()
	root.child = nil
	return root
}

func (this *FibonacciHeap) _unlink(node *Node) *Node {
	/*
	   \u0418\u0437\u0432\u043B\u0435\u0447\u0435\u043D\u0438\u0435 \u0443\u0437\u043B\u0430 \u0438\u0437 \u0434\u0432\u0443\u0445\u0441\u0432\u044F\u0437\u043D\u043E\u0433\u043E \u0441\u043F\u0438\u0441\u043A\u0430.

	   \u0412\u043E\u0437\u0432\u0440\u0430\u0449\u0430\u0435\u0442 \u043B\u0435\u0432\u044B\u0439 \u0443\u0437\u0435\u043B \u0438\u0437 \u043E\u0441\u0442\u0430\u0432\u0448\u0438\u0445\u0441\u044F \u0432 \u0441\u043F\u0438\u0441\u043A\u0435, \u043B\u0438\u0431\u043E None
	   left - node - right = left - right
	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(1)
	*/
	left := node.left
	right := node.right

	// \u0412 \u0441\u043F\u0438\u0441\u043A\u0435 1 \u044D\u043B\u0435\u043C\u0435\u043D\u0442 - \u0443\u0434\u0430\u043B\u044F\u0435\u043C\u044B\u0439
	if left == nil {
		return nil
	}

	if left == right {
		// \u0412 \u0441\u043F\u0438\u0441\u043A\u0435 \u0431\u044B\u043B\u043E 2 \u044D\u043B\u0435\u043C\u0435\u043D\u0442\u0430
		left.left = nil
		left.right = nil
	} else {
		left.right = right
		right.left = left
	}

	return left
}

func (this *FibonacciHeap) _consolidate() {
	/*
	   \u0423\u043F\u043B\u043E\u0442\u043D\u0435\u043D\u0438\u0435 \u0441\u043F\u0438\u0441\u043A\u0430 \u043A\u043E\u0440\u043D\u0435\u0439 - \u0441\u043A\u043B\u0435\u0438\u0432\u0430\u043D\u0438\u0435 \u0434\u0435\u0440\u0435\u0432\u044C\u0435\u0432 \u0441 \u043E\u0434\u0438\u043D\u0430\u043A\u043E\u0432\u044B\u043C \u0440\u0430\u043D\u0433\u043E\u043C.

	   \u041E\u0431\u043D\u043E\u0432\u043B\u044F\u0435\u0442 \u043C\u0438\u043D\u0438\u043C\u0430\u043B\u044C\u043D\u044B\u0439 \u0443\u0437\u0435\u043B
	   \u0438 \u0443\u0441\u0442\u0430\u043D\u0430\u0432\u043B\u0438\u0432\u0430\u0435\u0442 parent=None \u0434\u043B\u044F \u0432\u0441\u0435\u0445 \u043A\u043E\u0440\u043D\u0435\u0432\u044B\u0445 \u0443\u0437\u043B\u043E\u0432
	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(log n)
	*/
	// \u0432\u0440\u0435\u043C\u0435\u043D\u043D\u044B\u0439 \u043C\u0438\u043D\u0438\u043C\u0430\u043B\u044C\u043D\u044B\u0439 \u0443\u0437\u0435\u043B
	root := this.find_min()
	if root == nil {
		return
	}

	// \u0421\u043B\u043E\u0432\u0430\u0440\u044C \u043A\u043E\u0440\u043D\u0435\u0432\u044B\u0445 \u0443\u0437\u043B\u043E\u0432 \u0432\u0438\u0434\u0430 \u0440\u0430\u043D\u0433 -> \u0443\u0437\u0435\u043B
	ranked := map[int]*Node{}
	ranked[root.rank] = root
	root.parent = nil
	node := root.right

	for {
		if node == nil {
			break
		}
		// \u0423 \u043A\u043E\u0440\u043D\u044F \u043D\u0435\u0442 \u043F\u0440\u0435\u0434\u043A\u043E\u0432
		node.parent = nil
		// \u0422\u0435\u043A\u0443\u0449\u0438\u0439 \u0443\u0437\u0435\u043B
		melded := node
		// \u0421\u043B\u0435\u0434\u0443\u044E\u0449\u0438\u0439 \u043F\u0440\u043E\u0441\u043C\u0430\u0442\u0440\u0438\u0432\u0430\u0435\u043C\u044B\u0439 \u0443\u0437\u0435\u043B
		node = node.right
		v, ok := ranked[node.rank]
		if ok && v == node {
			// \u041C\u044B \u0442\u0430\u043C \u0443\u0436\u0435 \u0431\u044B\u043B\u0438, \u043F\u043E\u044D\u0442\u043E\u043C\u0443 \u044D\u0442\u0430 \u0438\u0442\u0435\u0440\u0430\u0446\u0438\u044F \u043F\u043E\u0441\u043B\u0435\u0434\u043D\u044F\u044F
			node = nil
		}

		for {
			// \u0412 \u0441\u043F\u0438\u0441\u043A\u0435 \u043A\u043E\u0440\u043D\u0435\u0439 \u0435\u0441\u0442\u044C \u0434\u0435\u0440\u0435\u0432\u043E \u0441 \u0442\u0430\u043A\u0438\u043C \u0436\u0435 \u0440\u0430\u043D\u0433\u043E\u043C.
			rank := melded.rank

			v, ok := ranked[rank]
			if !ok {
				break
			}
			// \u0421\u043A\u043B\u0435\u0438\u0432\u0430\u0435\u043C
			melded = this._link(melded, v)
			// \u0438 \u0443\u0434\u0430\u043B\u044F\u0435\u043C \u0438\u0437 \u0441\u043B\u043E\u0432\u0430\u0440\u044F \u043F\u0440\u0435\u0436\u043D\u0438\u0439 \u0440\u0430\u043D\u0433
			delete(ranked, rank)
		}
		// \u043E\u0431\u043D\u043E\u0432\u043B\u044F\u0435\u043C \u0441 \u043D\u043E\u0432\u044B\u043C \u0437\u043D\u0430\u0447\u0435\u043D\u0438\u0435\u043C \u0440\u0430\u043D\u0433\u0430 \u043F\u043E\u043B\u0443\u0447\u0438\u0432\u0448\u0435\u0435\u0441\u044F \u0434\u0435\u0440\u0435\u0432\u043E
		ranked[melded.rank] = melded
		// \u041E\u0431\u043D\u043E\u0432\u043B\u044F\u0435\u043C \u043C\u0438\u043D\u0438\u043C\u0430\u043B\u044C\u043D\u044B\u0439 \u0443\u0437\u0435\u043B
		this._update_min(melded)
	}
}

func (this *FibonacciHeap) _link(node1 *Node, node2 *Node) *Node {
	/*
	   \u0421\u043A\u043B\u0435\u0438\u0432\u0430\u043D\u0438\u0435 \u0434\u0432\u0443\u0445 \u043A\u043E\u0440\u043D\u0435\u0439.

	   \u041A\u043E\u0440\u043D\u0435\u043C \u0441\u0442\u0430\u043D\u043E\u0432\u0438\u0442\u0441\u044F \u0443\u0437\u0435\u043B \u0441 \u043C\u0435\u043D\u044C\u0448\u0438\u043C \u043A\u043B\u044E\u0447\u043E\u043C, \u0432\u0442\u043E\u0440\u043E\u0439 - \u0435\u0433\u043E \u043F\u043E\u0442\u043E\u043C\u043A\u043E\u043C
	   \u0412\u043E\u0437\u0432\u0440\u0430\u0449\u0430\u0435\u0442 \u043F\u043E\u043B\u0443\u0447\u0438\u0432\u0448\u0438\u0439\u0441\u044F \u043A\u043E\u0440\u0435\u043D\u044C
	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(1)
	*/
	if node1.key > node2.key {
		node1, node2 = node2, node1
	}
	// node1              node1
	//   |    ->            |
	// child      node2 - child

	// node2 \u0438\u0437\u0432\u043B\u0435\u043A\u0430\u0435\u0442\u0441\u044F \u0438\u0437 \u0441\u043F\u0438\u0441\u043A\u0430 \u043A\u043E\u0440\u043D\u0435\u0439
	this._unlink(node2)
	node2._extract()
	// \u0443\u0431\u0438\u0440\u0430\u0435\u0442\u0441\u044F \u043E\u0442\u043C\u0435\u0442\u043A\u0430
	node2.marked = false
	// \u0438 \u043E\u043D \u0441\u0442\u0430\u043D\u043E\u0432\u0438\u0442\u0441\u044F \u043F\u043E\u0442\u043E\u043C\u043A\u043E\u043C node1
	node2.parent = node1
	// \u041E\u0431\u043D\u043E\u0432\u043B\u044F\u0435\u043C \u0440\u0430\u043D\u0433 \u043F\u043E\u043B\u0443\u0447\u0438\u0432\u0448\u0435\u0433\u043E\u0441\u044F \u0434\u0435\u0440\u0435\u0432\u0430
	node1.rank += 1

	// \u041F\u043E\u0442\u043E\u043C\u043E\u043A \u043F\u0435\u0440\u0432\u043E\u0433\u043E \u043A\u043E\u0440\u043D\u044F
	child := node1.child
	if child == nil {
		// \u0415\u0441\u043B\u0438 \u043D\u0435\u0442 \u043F\u043E\u0442\u043E\u043C\u043A\u043E\u0432
		node1.child = node2
	} else {
		left := child.left
		if left == nil {
			// \u041E\u0434\u0438\u043D \u043F\u043E\u0442\u043E\u043C\u043E\u043A
			// child - node2
			child.left = node2
			child.right = node2
			node2.left = child
			node2.right = child
		} else {
			// left <-x child
			//   |        |
			//      node2
			node2.left = left
			node2.right = child
			left.right = node2
			child.left = node2
		}
	}

	return node1
}

func (this *FibonacciHeap) decrease_key(node *Node, newkey int) {
	/*
	   \u0423\u043C\u0435\u043D\u044C\u0448\u0435\u043D\u0438\u0435 \u043A\u043B\u044E\u0447\u0430 \u0443\u0437\u043B\u0430 node \u0434\u043E \u0437\u043D\u0430\u0447\u0435\u043D\u0438\u044F newkey.

	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(1)
	*/
	node.key = newkey

	if node.parent == nil {
		// \u0423\u0437\u0435\u043B - \u043A\u043E\u0440\u043D\u0435\u0432\u043E\u0439
		this._update_min(node)
		return
	}

	parent := node.parent
	parent.rank -= 1
	parent.child = this._unlink(node)
	this._cascading_cut(parent)
	node._extract()
	this.insert(node)
}

func (this *FibonacciHeap) _cut(node *Node) {
	/*
	   \u041F\u043E\u0434\u0440\u0435\u0437\u043A\u0430 \u0434\u0435\u0440\u0435\u0432\u0430 - \u043F\u0435\u0440\u0435\u043D\u043E\u0441 node \u0432 \u0441\u043F\u0438\u0441\u043E\u043A \u043A\u043E\u0440\u043D\u0435\u0439.

	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(1)
	*/
	parent := node.parent
	if parent == nil {
		// \u0423\u0437\u0435\u043B \u0443\u0436\u0435 \u043A\u043E\u0440\u043D\u0435\u0432\u043E\u0439
		return
	}
	parent.rank -= 1
	parent.child = this._unlink(node)
	node._extract()
	this.insert(node)
}

func (this *FibonacciHeap) _cascading_cut(node *Node) {
	/*
	   \u041A\u0430\u0441\u043A\u0430\u0434\u043D\u0430\u044F \u043F\u043E\u0434\u0440\u0435\u0437\u043A\u0430 \u0434\u0435\u0440\u0435\u0432\u0430.

	   \u041D\u0430\u0447\u0438\u043D\u0430\u044F \u043E\u0442 \u0443\u0437\u043B\u0430 node, \u0438 \u043F\u043E\u043A\u0430 \u043F\u0440\u0435\u0434\u0448\u0435\u0441\u0442\u0432\u0443\u044E\u0449\u0438\u0439 \u0443\u0437\u0435\u043B \u0438\u043C\u0435\u0435\u0442 \u043E\u0442\u043C\u0435\u0442\u043A\u0443
	   \u043E \u043F\u0435\u0440\u0435\u043C\u0435\u0449\u0435\u043D\u0438\u0438 (marked = True), \u0432\u0441\u0435 \u043E\u043D\u0438 \u0441\u0442\u0430\u043D\u043E\u0432\u044F\u0442\u0441\u044F \u043A\u043E\u0440\u043D\u0435\u0432\u044B\u043C\u0438.

	   \u0412\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(log n)
	*/
	parent := node
	for {
		if parent == nil {
			break
		}
		if !parent.marked {
			parent.marked = true
			return
		} else {
			node = parent
			parent = node.parent
			this._cut(node)
		}
	}
}

func (this *FibonacciHeap) remove(node *Node) *Node {
	/*
	   \u0423\u0434\u0430\u043B\u0435\u043D\u0438\u0435 \u0443\u0437\u043B\u0430 node.

	   \u0410\u043C\u043E\u0440\u0442\u0438\u0437\u0438\u0440\u043E\u0432\u0430\u043D\u043D\u043E\u0435 \u0432\u0440\u0435\u043C\u044F \u0440\u0430\u0431\u043E\u0442\u044B: O(log n)
	*/
	if node == this.find_min() {
		// \u0423\u0437\u0435\u043B - \u043C\u0438\u043D\u0438\u043C\u0430\u043B\u044C\u043D\u044B\u0439
		return this.delete_min()
	}

	parent := node.parent
	if parent == nil {
		// \u0423\u0437\u0435\u043B - \u043A\u043E\u0440\u043D\u0435\u0432\u043E\u0439
		this._unlink(node)
	} else {
		parent.rank -= 1
		parent.child = this._unlink(node)
		this._cascading_cut(parent)
	}

	h := ConstructHeap(node.child)
	this.meld(h)
	this._consolidate()
	node._extract()
	node.child = nil
	return node
}



type AuthenticationManager struct {
    fh *FibonacciHeap
    timeToLive int
    nodes map[string]*Node
}


func Constructor(timeToLive int) AuthenticationManager {
    this := AuthenticationManager{
        fh: ConstructHeap(nil),
        timeToLive: timeToLive,
        nodes: map[string]*Node{},
    }
    return this
}


func (this *AuthenticationManager) Generate(tokenId string, currentTime int)  {
    expired := currentTime + this.timeToLive
    node := ConstructNode(tokenId, expired)
    this.nodes[tokenId] = &node
    this.fh.insert(&node)
}


func (this *AuthenticationManager) Renew(tokenId string, currentTime int)  {
    node, ok := this.nodes[tokenId]
    if !ok {
        return
    }
    this.fh.remove(node)
    if node.key <= currentTime {
        delete(this.nodes, tokenId)
        return
    }
    expired := currentTime + this.timeToLive
    node.key = expired
    this.fh.insert(node)        
    
}


func (this *AuthenticationManager) CountUnexpiredTokens(currentTime int) int {
    for {
        node := this.fh.find_min()
        if node == nil {
            break
        }
        if node.key <= currentTime {
            this.fh.delete_min()
            delete(this.nodes, node.x)
        } else {
            break
        }
    }
    return len(this.nodes)
}
