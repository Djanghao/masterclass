class Solution {\r
\r
    /**\r
     * @param String $word1\r
     * @param String $word2\r
     * @return Boolean\r
     */\r
    function closeStrings($word1, $word2) {\r
        //\u041F\u0435\u0440\u0435\u0432\u043E\u0434\u0438\u043C \u0441\u043B\u043E\u0432\u0430 \u0432 \u043C\u0430\u0441\u0441\u0438\u0432\u044B \u0438 \u0441\u0440\u0430\u0432\u043D\u0438\u0432\u0430\u0435\u043C \u0438\u0445 \u0434\u043B\u0438\u043D\u043D\u0443\r
        $wordArr1 = str_split($word1);\r
        $wordArr2 = str_split($word2);\r
        //\u0415\u0441\u043B\u0438 \u0434\u043B\u0438\u043D\u043D\u0430 \u0440\u0430\u0437\u043D\u0430\u044F, \u0442\u043E \u0441\u0440\u0430\u0437\u0443 false\r
        if (count($wordArr1) !== count($wordArr2)) {\r
            return false;\r
        }\r
        //\u0424\u043E\u0440\u043C\u0438\u0440\u0443\u0435\u043C hash map \u0441 \u0431\u0443\u043A\u0432\u043E\u0439 \u0438 \u043A\u043E\u043B\u0438\u0447\u0435\u0441\u0442\u0432\u043E\u043C \u0435\u0435 \u043F\u043E\u0442\u043E\u0440\u0435\u043D\u0438\u0439 \u0432 \u0441\u043B\u043E\u0432\u0435\r
        $hash1 = array_count_values($wordArr1);\r
        $hash2 = array_count_values($wordArr2);\r
        //\u041F\u0440\u043E\u0432\u0435\u0440\u044F\u0435\u043C \u0447\u0442\u043E \u0432 \u043E\u0431\u043E\u0438\u0445 hash \u0441\u043E\u0434\u0435\u0440\u0436\u0430\u0442\u044C\u0441\u044F \u0441\u0442\u0440\u043E\u0433\u043E \u043E\u0434\u0438\u043D\u0430\u043A\u043E\u0432\u044B\u0435 \u0431\u0443\u043A\u0432\u044B, \u0445\u043E\u0442\u044F \u0431\u044B \u043F\u043E \u043E\u0434\u043D\u043E\u0439\r
        //\u0415\u0441\u043B\u0438 \u0432 \u043E\u0434\u043D\u043E\u043C \u0441\u043B\u043E\u0432\u0435 \u0435\u0441\u0442\u044C \u0431\u0443\u043A\u0432\u0430, \u043A\u043E\u0442\u043E\u0440\u043E\u0439 \u043D\u0435\u0442 \u0432 \u0434\u0440\u0443\u0433\u043E\u043C \u044D\u0442\u043E \u0441\u0440\u0430\u0437\u0443 false\r
        \r
        //\u041F\u0440\u043E\u0445\u043E\u0434\u0438\u043C \u043F\u043E \u043A\u0430\u0436\u0434\u043E\u043C\u0443 \u0437\u043D\u0430\u0447\u0435\u043D\u0438\u044E \u0438\u0437 \u043F\u0435\u0440\u0432\u043E\u0433\u043E hash\r
        foreach ($hash1 as $key => $value) {\r
            //\u0415\u0441\u043B\u0438 \u0442\u0430\u043A\u043E\u0433\u043E \u043A\u043B\u044E\u0447\u0430 \u043D\u0435\u0442 \u0432\u043E \u0432\u0442\u043E\u0440\u043E\u043C hash, \u0437\u043D\u0430\u0447\u0438\u0442 \u0432\u043E\u0437\u0432\u0440\u0430\u0449\u0430\u0435\u043C false (\u0432 \u0441\u043B\u043E\u0432\u0430\u0445 \u043F\u0440\u0438\u0441\u0443\u0442\u0441\u0442\u0432\u0443\u044E\u0442 \u0440\u0430\u0437\u043D\u044B\u0435 \u0431\u0443\u043A\u0432\u044B)\r
            if (!$hash2[$key]) {\r
                return false;\r
            }\r
        }\r
\r
        //\u041D\u0430\u043A\u043E\u043D\u0435\u0446 \u043F\u0440\u043E\u0432\u0435\u0440\u044F\u0435\u043C \u0447\u0442\u043E \u043A\u043E\u043B\u0438\u0447\u0435\u0441\u0442\u0432\u043E \u043F\u043E\u0432\u0442\u043E\u0440\u0435\u043D\u0438\u0439 \u0440\u0430\u0437\u043D\u044B\u0445 \u0431\u0443\u043A\u0432 \u043E\u0434\u0438\u043D\u0430\u043A\u043E\u0432\u043E\u0435\r
        //\u041D\u0430\u043F\u0440\u0438\u043C\u0435\u0440 \u0442\u0440\u0435\u0442\u0438\u0439 \u043F\u0440\u0438\u043C\u0435\u0440 cabbba \u0438 abbccc. Hash \u0431\u0443\u0434\u0443\u0442 \u0432\u044B\u0433\u043B\u044F\u0434\u0435\u0442\u044C \u043A\u0430\u043A\r
        //1) c - 1, a - 2, b - 3 2) c - 3, a - 1, b - 2\r
        //\u041D\u0443\u0436\u043D\u043E \u043F\u0440\u043E\u0432\u0435\u0440\u0438\u0442\u044C \u0447\u0442\u043E \u0438\u0437 1 2 3 \u043C\u043E\u0436\u043D\u043E \u043F\u043E\u043B\u0443\u0447\u0438\u0442\u044C 3 1 2. \u0422\u043E \u0435\u0441\u0442\u044C \u043D\u0435 \u0432\u0430\u0436\u043D\u044B \u0431\u0443\u043A\u0432\u044B, \u0430 \u0432\u0430\u0436\u043D\u043E \u0438\u0445 \u043A\u043E\u043B\u0438\u0447\u0435\u0441\u0442\u0432\u043E\r
        //\u0421\u043E\u0431\u0438\u0440\u0430\u0435\u043C \u043F\u043E \u043A\u0430\u0436\u0434\u043E\u043C\u0443 hash  \u0432 \u043C\u0430\u0441\u0441\u0438\u0432 \u0438 \u0441\u043E\u0440\u0442\u0438\u0440\u0443\u0435\u043C \u0435\u0433\u043E. \u0415\u0441\u043B\u0438 \u044D\u0442\u0438 \u043C\u0430\u0441\u0441\u0438\u0432\u044B \u0440\u0430\u0432\u043D\u044B \u0442\u043E \u0432\u043E\u0437\u0432\u0440\u0430\u0449\u0430\u0435\u043C true\r
        $nums1 = [];\r
        $nums2 = [];\r
        foreach ($hash1 as $value) {\r
            $nums1[] = $value;\r
        }\r
        foreach ($hash2 as $value) {\r
            $nums2[] = $value;\r
        }\r
\r
        //\u0421\u043E\u0440\u0442\u0438\u0440\u0443\u0435\u043C \u043C\u0430\u0441\u0441\u0438\u0432\u044B \u0438 \u0441\u0440\u0430\u0432\u043D\u0438\u0432\u0430\u0435\u043C\r
        sort($nums1);\r
        sort($nums2);\r
\r
        return $nums1 == $nums2 ? true : false;\r
    }\r
}\r
